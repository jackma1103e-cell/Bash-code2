#!/usr/bin/env bash
# Display the main menu options
display_menu() {
  echo ""
  echo "========================================"
  echo "       Enhanced Script Utility Menu     "
  echo "========================================"
  echo "1) Read contents from a file"
  echo "2) Copy contents from one file into another"
  echo "3) Write contents into a file (Overwrite)"
  echo "4) Compare 2 strings"
  echo "5) Search a file for a word"
  echo "6) Compare 2 numbers"
  echo "7) Count lines, words, and characters in a file"
  echo "8) Append text to a file"
  echo "0) Exit script"
  echo "========================================"
  echo ""
}

# 1) Read contents from a file
read_file() {
  read -rp "Enter the filename to read: " filename
  if [[ -f "$filename" ]]; then
    echo "--- File Content Start ---"
    cat "$filename"
    echo "--- File Content End ---"
  else
    echo "Error: File '$filename' does not exist."
  fi
}

# 2) Copy contents from one file into another
copy_file() {
  read -rp "Enter the source file to copy from: " sourceFile
  if [[ ! -f "$sourceFile" ]]; then
    echo "Error: Source file '$sourceFile' does not exist."
    return
  fi
  read -rp "Enter the destination file: " destFile
  cp "$sourceFile" "$destFile"
  echo "Successfully copied contents from '$sourceFile' to '$destFile'."
}

# 3) Write contents into a file (Overwrites existing contents)
write_file() {
  read -rp "Enter the filename to write to: " outputFile
  echo "Enter text to write to the file. Enter 'x' on a new line to finish."
  > "$outputFile"
  while true; do
    read -rp "> " line
    if [[ "$line" == "x" ]]; then
      echo "Exiting file write..."
      break
    fi
    echo "$line" >> "$outputFile"
  done
}

# 4) Compare 2 strings
compare_strings() {
  read -rp "Enter the first string: " str1
  read -rp "Enter the second string: " str2
  if [[ "$str1" == "$str2" ]]; then
    echo "Both strings are identical."
  else
    echo "The strings are different."
  fi
}

# 5) Search a file for a word
search_word() {
  read -rp "Enter the word/pattern to search for: " word
  read -rp "Enter the filename to search in: " filename
  if [[ ! -f "$filename" ]]; then
    echo "Error: File '$filename' does not exist."
    return
  fi
  grep -n --color=auto "$word" "$filename" || echo "No matches found."
}

# 6) Compare 2 numbers
compare_numbers() {
  read -rp "Enter the first number: " num1
  read -rp "Enter the second number: " num2
  if ((num1 == num2)); then
    echo "$num1 is equal to $num2"
  elif ((num1 > num2)); then
    echo "$num1 is greater than $num2"
  else
    echo "$num1 is less than $num2"
  fi
}

# Main Loop
option=-1
while [[ "$option" != "0" ]]; do
  display_menu
  read -rp "Select an option [0-8]: " option
  case "$option" in
    1) read_file ;;
    2) copy_file ;;
    3) write_file ;;
    4) compare_strings ;;
    5) search_word ;;
    6) compare_numbers ;;
    0) echo "Exiting script... Goodbye!" ;;
  esac
done
